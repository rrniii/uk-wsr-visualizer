cmake_minimum_required(VERSION 3.26)
project(UKWSRNative LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(HDF5_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(HDF5_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(HDF5_BUILD_HL_LIB OFF CACHE BOOL "" FORCE)
set(HDF5_BUILD_CPP_LIB OFF CACHE BOOL "" FORCE)
set(HDF5_BUILD_FORTRAN OFF CACHE BOOL "" FORCE)
set(HDF5_BUILD_JAVA OFF CACHE BOOL "" FORCE)
set(HDF5_ENABLE_ZLIB_SUPPORT ON CACHE BOOL "" FORCE)
set(HDF5_ENABLE_PLUGIN_SUPPORT OFF CACHE BOOL "" FORCE)
set(HDF5_ENABLE_THREADSAFE OFF CACHE BOOL "" FORCE)
set(HDF5_NO_PACKAGES ON CACHE BOOL "" FORCE)
set(HDF5_ENABLE_EMBEDDED_LIBINFO OFF CACHE BOOL "" FORCE)
set(HDF5_URL "https://github.com/HDFGroup/hdf5/archive/refs/tags/hdf5_2.1.0.tar.gz")
set(LOCAL_ARCHIVE "${CMAKE_CURRENT_LIST_DIR}/../../../../.cache/hdf5-2.1.0.tar.gz")
if(EXISTS "${LOCAL_ARCHIVE}")
    set(HDF5_URL "${LOCAL_ARCHIVE}")
endif()
FetchContent_Declare(hdf5 URL "${HDF5_URL}" URL_HASH SHA256=599f33c14ac1594df1541210b8bcc7b16459905380c784c381897ceafff18aff DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(hdf5)
set(SHARED_NATIVE "${CMAKE_CURRENT_LIST_DIR}/../../../../../shared/native")
add_library(ukwsr SHARED native_bridge.cpp "${SHARED_NATIVE}/UKHDF5Reader.c")
target_include_directories(ukwsr PRIVATE "${SHARED_NATIVE}" "${hdf5_SOURCE_DIR}/src" "${hdf5_BINARY_DIR}/src")
target_link_libraries(ukwsr PRIVATE hdf5-static log z)
target_compile_features(ukwsr PRIVATE cxx_std_17)
target_link_options(ukwsr PRIVATE "-Wl,-z,max-page-size=16384")
